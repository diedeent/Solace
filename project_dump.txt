#pragma once

#include <ozz/animation/runtime/animation.h>

#include <string>
#include <vector>

namespace Solace {

    // ============================================================
    // FrameEvent — событие внутри клипа, триггерится при проходе кадра.
    //
    // frame считается по FPS клипа (обычно 30). При dt = 0.016 (60 FPS)
    // один игровой кадр = ~0.5 кадра анимации.
    //
    // Пример: attack_sword (30 FPS, длина 24 кадра):
    //   frame 8  → "swing" (замах)
    //   frame 14 → "hit"   (hitbox active start)
    //   frame 18 → "hit_end" (hitbox active end)
    // ============================================================
    struct FrameEvent {
        int         frame;
        std::string name;
    };


    // ============================================================
    // AnimationClip — runtime animation + metadata.
    //
    // - animation    : ozz::animation::Animation (shared, из ModelAsset)
    // - name         : "idle", "run", "attack_1", "death"
    // - fps          : frame rate для events (30 обычно)
    // - loop         : зациклен? (idle/run = true, attack/death = false)
    // - events       : frame-triggered events
    // - interrupts   : какие клипы можно прервать этим клипом (для ASM)
    // ============================================================
    struct AnimationClip {
        const ozz::animation::Animation* animation = nullptr;
        std::string name;
        float fps = 30.0f;
        bool  loop = true;
        std::vector<FrameEvent> events;

        float duration() const {
            return animation ? animation->duration() : 0.0f;
        }

        int total_frames() const {
            return static_cast<int>(duration() * fps);
        }

        int frame_at(float time) const {
            return static_cast<int>(time * fps);
        }
    };

} // namespace Solace

#pragma once

#include <algorithm>
#include <cstdint>
#include <chrono>

namespace Solace {

    // ============================================================
    // Clock — управление временем для gameplay и UI.
    //
    // Три времени:
    //   - realDt:      реальное время кадра (clamp [1ms, 100ms])
    //   - gameplayDt:  realDt * scale, НО = 0 во время hitstop или pause
    //
    // HitStop: request_hitstop(duration) замораживает gameplayDt
    //          на `duration` секунд REAL-времени.
    //
    // Проблема, которую решает: при hitstop UI/input/camera shake
    // не должны замедляться. Только gameplay.
    // ============================================================
    class Clock {
    public:
        Clock() = default;
        ~Clock() = default;

        Clock(const Clock&) = delete;
        Clock& operator=(const Clock&) = delete;
        Clock(Clock&&) = delete;
        Clock& operator=(Clock&&) = delete;

        // ============ Жизненный цикл кадра ============

        // Вызывается ПЕРВЫМ в Application::run()
        void tick() {
            const uint64_t now = now_ns();

            if (m_last_ns == 0) {
                // Первый кадр — delta не определена
                m_real_dt = 0.0f;
                m_last_ns = now;
                update_gameplay_dt();
                return;
            }

            const uint64_t diff_ns = now - m_last_ns;
            m_last_ns = now;

            // Clamp: [1ms, 100ms]
            // Минимум — защита от деления на ~0 в FPS
            // Максимум — защита от spiral of death при лагах
            const float raw_dt = static_cast<float>(diff_ns) * 1e-9f;
            m_real_dt = std::clamp(raw_dt, MIN_DT, MAX_DT);

            // HitStop отсчитывается по REAL-времени
            if (m_hitstop_remaining > 0.0f) {
                m_hitstop_remaining -= m_real_dt;
                if (m_hitstop_remaining < 0.0f) m_hitstop_remaining = 0.0f;
            }

            m_total_real += m_real_dt;
            update_gameplay_dt();
        }

        // Вызывается ПОСЛЕДНИМ (после render, UI)
        void frame_end() {
            m_frame_count++;

            // FPS: обновляем раз в секунду real-времени
            m_fps_accum += m_real_dt;
            if (m_fps_accum >= 1.0f) {
                m_fps = static_cast<float>(m_frame_count) / m_fps_accum;
                m_fps_accum = 0.0f;
                m_frame_count = 0;
            }
        }

        // ============ Доступ к времени ============

        float real_dt() const { return m_real_dt; }
        float gameplay_dt() const { return m_gameplay_dt; }
        float total_real() const { return m_total_real; }
        float total_gameplay() const { return m_total_gameplay; }

        // ============ Управление временем ============

        // Пауза: gameplay замирает, real идёт
        void set_paused(bool paused) { m_paused = paused; }
        bool is_paused() const { return m_paused; }

        // Slow-mo (для будущих боссов / спецэффектов)
        // 1.0 = норма, 0.5 = в 2 раза медленнее, 0.0 = стоп
        void set_gameplay_scale(float scale) {
            m_gameplay_scale = std::clamp(scale, 0.0f, 10.0f);
        }
        float gameplay_scale() const { return m_gameplay_scale; }

        // HitStop: замораживает gameplayDt на duration REAL-секунд.
        // Если вызвать повторно — берётся max (не суммируется).
        // Если hitstop уже активен — новый заменяет старый, если длиннее.
        void request_hitstop(float duration) {
            if (duration <= 0.0f) return;
            m_hitstop_remaining = std::max(m_hitstop_remaining, duration);
        }

        bool is_in_hitstop() const { return m_hitstop_remaining > 0.0f; }

        // ============ Debug ============

        float fps() const { return m_fps; }

        struct Stats {
            float real_dt;
            float gameplay_dt;
            float total_real;
            float total_gameplay;
            float fps;
            bool paused;
            bool in_hitstop;
            float hitstop_remaining;
            float gameplay_scale;
        };

        Stats debug_stats() const {
            return Stats{
                m_real_dt, m_gameplay_dt, m_total_real, m_total_gameplay,
                m_fps, m_paused, is_in_hitstop(), m_hitstop_remaining, m_gameplay_scale
            };
        }

        // ============ Сброс (при смене сцены) ============
        void reset() {
            m_last_ns = 0;
            m_real_dt = 0.0f;
            m_gameplay_dt = 0.0f;
            m_total_real = 0.0f;
            m_total_gameplay = 0.0f;
            m_hitstop_remaining = 0.0f;
            m_paused = false;
            m_gameplay_scale = 1.0f;
            m_fps = 0.0f;
            m_fps_accum = 0.0f;
            m_frame_count = 0;
        }

    private:
        static constexpr float MIN_DT = 0.001f;  // 1 ms (1000 FPS максимум)
        static constexpr float MAX_DT = 0.100f;  // 100 ms (10 FPS минимум)

        void update_gameplay_dt() {
            if (m_paused || is_in_hitstop()) {
                m_gameplay_dt = 0.0f;
            }
            else {
                m_gameplay_dt = m_real_dt * m_gameplay_scale;
            }
            m_total_gameplay += m_gameplay_dt;
        }

        static uint64_t now_ns() {
            // В будущем — вынести в Platform layer.
            // Пока используем std::chrono, чтобы Clock не зависел от SDL.
            return static_cast<uint64_t>(
                std::chrono::high_resolution_clock::now().time_since_epoch().count()
                );
        }

        uint64_t m_last_ns = 0;
        float m_real_dt = 0.0f;
        float m_gameplay_dt = 0.0f;
        float m_total_real = 0.0f;
        float m_total_gameplay = 0.0f;

        float m_hitstop_remaining = 0.0f;
        bool  m_paused = false;
        float m_gameplay_scale = 1.0f;

        // FPS counter (встроен, чтобы не плодить сущности)
        float m_fps = 0.0f;
        float m_fps_accum = 0.0f;
        int   m_frame_count = 0;
    };

} // namespace Solace

#pragma once

#include <engine/core/event_bus.hpp>
#include <engine/core/logger.hpp>

#include <nlohmann/json.hpp>

#include <cassert>
#include <filesystem>
#include <fstream>
#include <functional>
#include <memory>
#include <string>
#include <typeindex>
#include <unordered_map>

namespace Solace {

    struct ConfigReloadRequestEvent {};
    struct ConfigReloadedEvent { bool success = false; };

    class ConfigCache {
    public:
        ConfigCache() = default;
        ~ConfigCache() = default;
        ConfigCache(const ConfigCache&) = delete;
        ConfigCache& operator=(const ConfigCache&) = delete;

        template<typename T>
        using Parser = std::function<bool(const nlohmann::json&, T&)>;

        template<typename T>
        void register_section(std::string name, Parser<T> parser) {
            auto section = std::make_unique<Section<T>>();
            section->parser = std::move(parser);
            m_type_to_name[std::type_index(typeid(T))] = name;
            m_sections.emplace(std::move(name), std::move(section));
        }

        bool load(const std::string& root_dir) {
            m_root_dir = root_dir;
            return reload();
        }

        bool reload() {
            if (m_root_dir.empty()) {
                Logger::warning("ConfigCache: reload() before load()");
                return false;
            }

            bool all_ok = true;
            for (auto& [name, section] : m_sections) {
                if (!reload_section(name, *section)) all_ok = false;
            }

            Logger::info(all_ok ? "ConfigCache: all sections reloaded"
                : "ConfigCache: some sections failed");
            return all_ok;
        }

        template<typename T>
        const T& get() const {
            const auto* section = find_section<T>();
            assert(section && "ConfigCache::get<T>() — type not registered");
            return section->data;
        }

        std::vector<std::string> debug_section_names() const {
            std::vector<std::string> names;
            names.reserve(m_sections.size());
            for (const auto& [name, _] : m_sections) names.push_back(name);
            return names;

        bool has() const { return find_section<T>() != nullptr; }

    private:

            virtual ~SectionBase() = default;
            virtual bool parse_from_json(const nlohmann::json& j) = 0;

        template<typename T>
        struct Section : SectionBase {

            Parser<T> parser;

            bool parse_from_json(const nlohmann::json& j) override {
                T temp = data;  // откат при ошибке парсинга
                try {
                    if (parser(j, temp)) {
                        data = std::move(temp);
                        return true;
                    }
                }
                catch (const std::exception& e) {
                    Logger::error("ConfigCache: parse exception: ", e.what());
                }
                return false;  // data остаётся прежней
            }
        };

        template<typename T>
        const Section<T>* find_section() const {
            auto it = m_type_to_name.find(std::type_index(typeid(T)));
            if (it == m_type_to_name.end()) return nullptr;
            auto sec_it = m_sections.find(it->second);
            if (sec_it == m_sections.end()) return nullptr;
            return static_cast<const Section<T>*>(sec_it->second.get());
        }

        bool reload_section(const std::string& name, SectionBase& section) {

            }
            catch (const std::exception& e) {
                Logger::error("ConfigCache: JSON error in ", file.string(), ": ", e.what());
                return false;
            }
        }

        std::unordered_map<std::type_index, std::string> m_type_to_name;

    // ============================================================
    // ComponentPool<T> — плотный массив компонентов, индексированный EntityID.
    //
    // - vector<T>        : данные (index = EntityID)
    // - vector<bool>     : присутствие (index = EntityID)
    //
    // Гарантии:
    //   - get(id) возвращает nullptr, если компонента нет или id мёртв
    //   - add(id) перезаписывает существующий компонент
    //   - remove(id) не ломает итерацию (только флаг)
    //   - for_each итерирует только живые компоненты

    template<typename T>
    class ComponentPool {
    public:
        ComponentPool() = default;

        T* add(EntityID id, T component) {
            if (id == INVALID_ENTITY || id >= MAX_ENTITIES) return nullptr;
            ensure_size(id + 1);
            m_data[id] = std::move(component);
            m_present[id] = true;
            return &m_data[id];
        }

        T* get(EntityID id) {
            if (id == INVALID_ENTITY || id >= m_data.size()) return nullptr;
            if (!m_present[id]) return nullptr;
            return &m_data[id];
        }

        const T* get(EntityID id) const {
            if (id == INVALID_ENTITY || id >= m_data.size()) return nullptr;
            if (!m_present[id]) return nullptr;
            return &m_data[id];
        }

        bool has(EntityID id) const {
            return get(id) != nullptr;
        }

        void remove(EntityID id) {
            if (id == INVALID_ENTITY || id >= m_data.size()) return;
            m_present[id] = false;
        }

        template<typename Fn>
        void for_each(Fn&& fn) {
            for (size_t i = 1; i < m_data.size(); ++i) {
                if (m_present[i]) fn(static_cast<EntityID>(i), m_data[i]);
            }
        }

        template<typename Fn>
        void for_each(Fn&& fn) const {
            for (size_t i = 1; i < m_data.size(); ++i) {
                if (m_present[i]) fn(static_cast<EntityID>(i), m_data[i]);
            }
        }

        size_t count() const {
            size_t n = 0;
            for (size_t i = 1; i < m_present.size(); ++i) n += m_present[i] ? 1 : 0;
            return n;
        }

        void clear() {
            m_data.clear();
            m_present.clear();
        }

    private:
        void ensure_size(size_t new_size) {
            if (new_size > m_data.size()) {
                m_data.resize(new_size);
                m_present.resize(new_size, false);
            }

        std::vector<T>    m_data;
        std::vector<bool> m_present;

    // EntityRegistry — владелец сущностей и их компонентов.
    //
    // Гарантии:
    //   - create() → новый EntityID (никогда не переиспользуется)
    //   - destroy(id) → deferred, is_alive(id) = false сразу
    //   - process_destroy() → удаляет компоненты + emit EntityDestroyedEvent
    //   - get_component<T>(id) → nullptr, если id мёртв/нет компонента
    //   - for_each<A, B>() итерирует только сущности с A и B
    // ============================================================
    class EntityRegistry {
    public:
        explicit EntityRegistry(EventBus& bus) : m_bus(bus) {}
        ~EntityRegistry() = default;

        EntityRegistry(const EntityRegistry&) = delete;
        EntityRegistry& operator=(const EntityRegistry&) = delete;

        // ============ Жизненный цикл ============

        EntityID create() {
            if (m_next_id >= MAX_ENTITIES) {
                // 64k за сессию — не должно случаться. Если случилось — баг.
                return INVALID_ENTITY;
            }
            const EntityID id = m_next_id++;
            m_alive[id] = true;
            return id;
        }

        void destroy(EntityID id) {
            if (!is_alive(id)) return;
            m_alive[id] = false;
            m_pending_destroy.push_back(id);
        }

        void process_destroy() {
            if (m_pending_destroy.empty()) return;

            for (EntityID id : m_pending_destroy) {
                // Удаляем все компоненты
                for (auto& [_, pool] : m_pools) {
                    (void)_;
                    // Каждый pool знает свой T — нужен type-erased remove.
                    // Используем visitor-подход через лямбду, сохранённую в pool.
                    // См. PoolBase::remove_by_id
                }
                // Эмитим событие
                m_bus.emit(EntityDestroyedEvent{ id });
            }
            m_pending_destroy.clear();
        }

            if (id == INVALID_ENTITY || id >= m_alive.size()) return false;
            return m_alive[id];

        template<typename T>
        T* add_component(EntityID id, T component) {
            if (!is_alive(id)) return nullptr;

        bool has_component(EntityID id) const {
            if (!is_alive(id)) return false;
            auto* pool = find_pool<T>();
            return pool && pool->has(id);
        }

        template<typename T>
        void remove_component(EntityID id) {
            if (!is_alive(id)) return;
            auto* pool = find_pool<T>();
            if (pool) pool->remove(id);

        template<typename... Components, typename Fn>
        void for_each(Fn&& fn) {
            // Итерируем по первой компоненте, проверяем остальные
            if constexpr (sizeof...(Components) == 0) {
                // Только по сущностям
                for (size_t i = 1; i < m_alive.size(); ++i) {
                    if (m_alive[i]) fn(static_cast<EntityID>(i));
                }
            }
            else if constexpr (sizeof...(Components) == 1) {
                using First = std::tuple_element_t<0, std::tuple<Components...>>;
                auto* pool = find_pool<First>();
                if (!pool) return;
                pool->for_each([&](EntityID id, First& c) {
                    fn(id, c);
                    });

                auto* first_pool = find_pool<First>();
                if (!first_pool) return;
                first_pool->for_each([&](EntityID id, First& c0) {
                    if (has_all<Components...>(id)) {
                        invoke_with_all<Components...>(id, [&](auto&... comps) {
                            fn(id, comps...);
                            });
                    }
                    });
            }
        }

            for (auto& [_, pool] : m_pools) (void)_;
            // Очищаем все pools через их clear()
            for (auto& [_, pool] : m_pools) {
                (void)_;
                // Нужно вызвать clear() на PoolBase — см. PoolBase::clear
            }
            m_pools.clear();
            m_alive.clear();
            m_pending_destroy.clear();
            m_next_id = 1;
        }

        size_t alive_count() const {
            size_t n = 0;
            for (size_t i = 1; i < m_alive.size(); ++i) n += m_alive[i] ? 1 : 0;
            return n;
        }

        size_t pending_count() const { return m_pending_destroy.size(); }

        EntityID next_id() const { return m_next_id; }

    private:
        // ---- Type-erased base для всех pools ----
        struct PoolBase {
            virtual ~PoolBase() = default;
            virtual void clear() = 0;
            virtual void remove_by_id(EntityID id) = 0;
        };

        template<typename T>
        struct PoolWrapper : PoolBase {
            ComponentPool<T> pool;
            void clear() override { pool.clear(); }
            void remove_by_id(EntityID id) override { pool.remove(id); }
        };

        template<typename T>
        PoolWrapper<T>& get_or_create_pool() {
            auto typeId = std::type_index(typeid(T));
            auto it = m_pools.find(typeId);
            if (it == m_pools.end()) {
                auto [new_it, _] = m_pools.emplace(typeId, std::make_unique<PoolWrapper<T>>());
                return *static_cast<PoolWrapper<T>*>(new_it->second.get());
            }
            return *static_cast<PoolWrapper<T>*>(it->second.get());
        }

        template<typename T>
        PoolWrapper<T>* find_pool() {
            auto it = m_pools.find(std::type_index(typeid(T)));
            if (it == m_pools.end()) return nullptr;
            return static_cast<PoolWrapper<T>*>(it->second.get());
        }

        template<typename T>
        const PoolWrapper<T>* find_pool() const {
            auto it = m_pools.find(std::type_index(typeid(T)));
            if (it == m_pools.end()) return nullptr;
            return static_cast<const PoolWrapper<T>*>(it->second.get());
        }

        template<typename... Components>
        bool has_all(EntityID id) {
            bool all = true;
            ((all = all && has_component<Components>(id)), ...);
            return all;
        }

        template<typename... Components, typename Fn>
        void invoke_with_all(EntityID id, Fn&& fn) {
            fn(*get_component<Components>(id)...);
        }

        EntityID m_next_id = 1;
        std::vector<bool> m_alive;
        std::vector<EntityID> m_pending_destroy;
        std::unordered_map<std::type_index, std::unique_ptr<PoolBase>> m_pools;
    };

namespace Solace {

    // 0 зарезервирован как INVALID
    using SubscriptionID = uint64_t;
    constexpr SubscriptionID INVALID_SUBSCRIPTION = 0;

    class ScopedSubscriptions;

    // ============================================================
    // EventBus — type-erased, DI-driven pub/sub
    //
    // Жизненный цикл: Application владеет EventBus, передаёт по &
    // в Scene, Scene — в системы через конструктор.
    //
    // Гарантии:
    //   - Нет static/singleton
    //   - emit = sync, queue = deferred (FIFO, в конце кадра)
    //   - unsubscribe безопасен во время emit (snapshot)
    //   - ID не переиспользуются даже после clear()
    //   - ScopedSubscriptions автоматически отписывают при уничтожении
    //   - ~EventBus инвалидирует все ScopedSubscriptions
    // ============================================================
    class EventBus {
    public:
        EventBus() = default;
        ~EventBus();

        EventBus(const EventBus&) = delete;
        EventBus& operator=(const EventBus&) = delete;
        EventBus(EventBus&&) = delete;              // ссылки в Scoped не выживут move
        EventBus& operator=(EventBus&&) = delete;

        template<typename T>
        using EventCallback = std::function<void(const T&)>;

        // -------- Подписка / отписка --------

        template<typename T>
        SubscriptionID subscribe(EventCallback<T> callback) {
            const auto typeId = std::type_index(typeid(T));
            const SubscriptionID id = m_nextSubscriptionID++;

            ErasedHandler erased = [cb = std::move(callback)](const void* data) {
                cb(*static_cast<const T*>(data));
                };

            m_subscribers[typeId].push_back(Subscription{ id, std::move(erased) });
            m_idToType.emplace(id, typeId);
            return id;
        }

        void unsubscribe(SubscriptionID id) {
            auto it = m_idToType.find(id);
            if (it == m_idToType.end()) return;

            const std::type_index typeId = it->second;
            auto bucketIt = m_subscribers.find(typeId);
            if (bucketIt != m_subscribers.end()) {
                auto& list = bucketIt->second;
                list.erase(
                    std::remove_if(list.begin(), list.end(),
                        [id](const Subscription& s) { return s.id == id; }),
                    list.end()
                );
            }
            m_idToType.erase(it);
        }

        // -------- Эмиссия --------

        template<typename T>
        void emit(const T& event) {
#ifndef NDEBUG
            // Guard от рекурсии: emit внутри emit допустим только для
            // РАЗНЫХ типов. Защита от бесконечного цикла — на совести
            // подписчика (см. Handbook 1.3).
            assert(!m_inEmitOfSameType(typeid(T)) &&
                "emit<SameType> called re-entrantly — infinite loop risk");
            m_emitStack.push_back(typeid(T));
#endif

            const auto typeId = std::type_index(typeid(T));
            auto it = m_subscribers.find(typeId);
            if (it != m_subscribers.end()) {
                // Snapshot: если handler вызывает subscribe/unsubscribe,
                // итерация не сломается.
                auto snapshot = it->second;
                for (const auto& sub : snapshot) {
                    sub.handler(&event);
                }
            }

#ifndef NDEBUG
            m_emitStack.pop_back();
#endif
        }

        template<typename T>
        void queue(T event) {
            // Захватываем event по значению в замыкание — тип сохраняется.
            m_queue.push([this, ev = std::move(event)]() mutable {
                emit(ev);
                });
        }

        void processQueue() {
            // Snapshot + swap: если во время обработки кто-то queue'ит новое,
            // оно попадёт в СЛЕДУЮЩИЙ кадр (не в этот).
            auto pending = std::move(m_queue);
            m_queue = {};

            while (!pending.empty()) {
                pending.front()();
                pending.pop();
            }
        }

        // -------- Управление --------

        void clear() {
            m_subscribers.clear();
            m_idToType.clear();
            m_queue = {};
            // m_nextSubscriptionID НЕ сбрасываем — иначе коллизия ID
            // с живыми ScopedSubscriptions.
        }

        template<typename T>
        bool has_subscribers() const {
            auto it = m_subscribers.find(std::type_index(typeid(T)));
            return it != m_subscribers.end() && !it->second.empty();
        }

        // Debug
        struct Stats {
            size_t total_subscribers = 0;
            size_t unique_event_types = 0;
            size_t queued_events = 0;
        };
        Stats debug_stats() const {
            Stats s;
            s.unique_event_types = m_subscribers.size();
            s.queued_events = m_queue.size();
            for (const auto& [_, list] : m_subscribers) s.total_subscribers += list.size();
            return s;
        }

    private:
        friend class ScopedSubscriptions;

        using ErasedHandler = std::function<void(const void*)>;

        struct Subscription {
            SubscriptionID id;
            ErasedHandler handler;
        };

        std::unordered_map<std::type_index, std::vector<Subscription>> m_subscribers;
        std::unordered_map<SubscriptionID, std::type_index> m_idToType;
        std::queue<std::function<void()>> m_queue;
        SubscriptionID m_nextSubscriptionID = 1;  // 0 = INVALID

        // Список живых ScopedSubscriptions для инвалидации при ~EventBus
        std::unordered_set<ScopedSubscriptions*> m_scoped;

        void invalidate_all_scoped();

#ifndef NDEBUG
        std::vector<std::type_index> m_emitStack;
        bool m_inEmitOfSameType(const std::type_info& t) const {
            const auto idx = std::type_index(t);
            return std::find(m_emitStack.begin(), m_emitStack.end(), idx) != m_emitStack.end();
        }
#endif
    };


    // ============================================================
    // ScopedSubscriptions — RAII обёртка над набором подписок.
    //
    // Гарантии:
    //   - При уничтожении отписывает всё, что подписала
    //   - Если EventBus уничтожен раньше — безопасно (m_bus = nullptr)
    //   - Move-safe: перенос владения
    // ============================================================
    class ScopedSubscriptions {
    public:
        explicit ScopedSubscriptions(EventBus& bus) : m_bus(&bus) {
            bus.m_scoped.insert(this);
        }

        ~ScopedSubscriptions() {
            clear();
            if (m_bus) m_bus->m_scoped.erase(this);
        }

        ScopedSubscriptions(const ScopedSubscriptions&) = delete;
        ScopedSubscriptions& operator=(const ScopedSubscriptions&) = delete;

        ScopedSubscriptions(ScopedSubscriptions&& other) noexcept
            : m_bus(other.m_bus), m_ids(std::move(other.m_ids)) {
            if (m_bus) {
                m_bus->m_scoped.erase(&other);
                m_bus->m_scoped.insert(this);
            }
            other.m_bus = nullptr;
            other.m_ids.clear();
        }

        ScopedSubscriptions& operator=(ScopedSubscriptions&& other) noexcept {
            if (this != &other) {
                clear();
                if (m_bus) m_bus->m_scoped.erase(this);

                m_bus = other.m_bus;
                m_ids = std::move(other.m_ids);

                if (m_bus) {
                    m_bus->m_scoped.erase(&other);
                    m_bus->m_scoped.insert(this);
                }
                other.m_bus = nullptr;
                other.m_ids.clear();
            }
            return *this;
        }

        template<typename T>
        void subscribe(EventBus::EventCallback<T> callback) {
            if (!m_bus) return;
            m_ids.push_back(m_bus->subscribe<T>(std::move(callback)));
        }

        void clear() {
            if (!m_bus) {
                m_ids.clear();
                return;
            }
            for (auto id : m_ids) m_bus->unsubscribe(id);
            m_ids.clear();
        }

    private:
        friend class EventBus;
        EventBus* m_bus = nullptr;
        std::vector<SubscriptionID> m_ids;
    };

    inline EventBus::~EventBus() {
        invalidate_all_scoped();
    }

    inline void EventBus::invalidate_all_scoped() {
        for (auto* scoped : m_scoped) {
            scoped->m_bus = nullptr;
            scoped->m_ids.clear();
        }
        m_scoped.clear();
    }

#pragma once

#include <iostream>
#include <string_view>

namespace Solace {
    class Logger {
    public:
        // Информационное сообщение (Зеленый цвет)
        template<typename... Args>
        static void info(Args&&... args) {
            std::cout << "\033[32m[INFO]\033[0m ";
            print(std::forward<Args>(args)...);
        }

        // Предупреждение (Желтый цвет)
        template<typename... Args>
        static void warning(Args&&... args) {
            std::cout << "\033[33m[WARNING]\033[0m ";
            print(std::forward<Args>(args)...);
        }

        // Критическая ошибка (Красный цвет)
        template<typename... Args>
        static void error(Args&&... args) {
            std::cerr << "\033[31m[ERROR]\033[0m ";
            print(std::forward<Args>(args)...);
        }

        // Отладочное сообщение (Голубой/Циан цвет)
        template<typename... Args>
        static void debug(Args&&... args) {
            std::cout << "\033[36m[DEBUG]\033[0m ";
            print(std::forward<Args>(args)...);
        }

    private:
        // Вспомогательная функция для склеивания любых аргументов в одну строчку (фишка C++17/C++20)
        template<typename... Args>
        static void print(Args&&... args) {
            (std::cout << ... << args);
            std::cout << "\n";
        }
    };
}

    // ============================================================
    // SceneContext — shared services для всех сцен.
    //
    // Создаётся один раз в Engine, передаётся по & в Scene::on_*.
    // Reference-члены → non-copyable, non-movable.
    //
    // Layer: 5 (знает про 0-4, но не про конкретные сцены).
    // ============================================================
    struct SceneContext {
        EventBus& bus;
        Clock& clock;
        ConfigCache& config;
        EntityRegistry& registry;
        AssetCache& assets;

        // Legacy (Phase 0.4) — graphics через SettingsManager
        // Phase 0.7 мигрирует на ConfigCache
        class Window& window;
        class PostProcess& post_process;

        SceneContext(
            EventBus& b,
            Clock& c,
            ConfigCache& cfg,
            EntityRegistry& reg,
            AssetCache& a,
            Window& w,
            PostProcess& pp
        ) : bus(b), clock(c), config(cfg), registry(reg),
            assets(a), window(w), post_process(pp) {
        }

        SceneContext(const SceneContext&) = delete;
        SceneContext& operator=(const SceneContext&) = delete;
    };


    // ============================================================
    // SceneTransition — отложенные команды SceneManager.
    // ============================================================
    enum class SceneTransitionType {
        None,
        Push,
        Pop,
        Replace,
    };


    // ============================================================
    // Scene — интерфейс жизненного цикла.
    //
    // Контракт:
    //   - on_enter(ctx):    сцена получает ctx, инициализирует ресурсы
    //   - on_update:        системы обновляются
    //   - on_render:        рисует в низкое разрешение (PostProcess уже активен)
    //   - on_ui_render:     ImGui поверх (не обязательно)
    //   - on_event:         SDL_Event (до InputSystem)
    //   - on_exit(ctx):     ОБЯЗАН освободить все ресурсы (registry.clear,
    //                       ScopedSubscriptions clear)
    //
    // Сцена НЕ должна:
    //   - вызывать SceneManager напрямую (только через request_*)
    //   - знать про другие сцены
    // ============================================================
    class Scene {
    public:
        virtual ~Scene() = default;

        virtual void on_enter(SceneContext& ctx) = 0;
        virtual void on_exit(SceneContext& ctx) = 0;

        virtual void on_update(float gameplay_dt, float real_dt, SceneContext& ctx) = 0;
        virtual void on_render(SceneContext& ctx) = 0;
        virtual void on_ui_render(SceneContext& ctx) {}
        virtual void on_event(const SDL_Event& event, SceneContext& ctx) {}

        // Отложенные переходы — SceneManager применит в process_transitions()
        void request_push(std::unique_ptr<Scene> next);
        void request_pop();
        void request_replace(std::unique_ptr<Scene> next);

    protected:
        Scene() = default;

    private:
        friend class SceneManager;

        SceneTransitionType m_pending_type = SceneTransitionType::None;
        std::unique_ptr<Scene> m_pending_scene;
    };

} // namespace Solace

#pragma once

#include <engine/core/logger.hpp>
#include <engine/scene/scene.hpp>

#include <memory>
#include <vector>

namespace Solace {

    // ============================================================
    // SceneManager — стек сцен с отложенными переходами.
    //
    // Гарантии:
    //   - В любой момент активна только верхняя сцена
    //   - request_push/pop/replace применяются в process_transitions()
    //   - on_exit старой вызывается ДО on_enter новой
    //   - Если стек пустой — update/render безопасны (no-op + warning)
    // ============================================================
    class SceneManager {
    public:
        explicit SceneManager(SceneContext& ctx) : m_ctx(ctx) {}
        ~SceneManager() {
            // Уничтожаем сцены сверху вниз (обратный порядок стек)
            while (!m_stack.empty()) {
                m_stack.back()->on_exit(m_ctx);
                m_stack.pop_back();
            }
        }

        SceneManager(const SceneManager&) = delete;
        SceneManager& operator=(const SceneManager&) = delete;

        // Немедленный push (только в начальной инициализации)
        void push_immediate(std::unique_ptr<Scene> scene) {
            if (!scene) return;
            Logger::info("SceneManager: push_immediate");
            scene->on_enter(m_ctx);
            m_stack.push_back(std::move(scene));
        }

        // ---- Применение отложенных переходов ----
        void process_transitions() {
            // Собираем pending команды со всех сцен в стеке
            // (обычно только с верхней, но on_event может ставить на любую)
            for (auto& scene : m_stack) {
                if (scene->m_pending_type != SceneTransitionType::None) {
                    apply_transition(*scene);
                    return;  // один переход за кадр — проще и предсказуемо
                }
            }
        }

        // ---- Lifecycle ----
        void update(float gameplay_dt, float real_dt) {
            if (m_stack.empty()) {
                Logger::warning("SceneManager: update with empty stack");
                return;
            }
            m_stack.back()->on_update(gameplay_dt, real_dt, m_ctx);
        }

        void render() {
            if (m_stack.empty()) return;
            m_stack.back()->on_render(m_ctx);
        }

        void ui_render() {
            if (m_stack.empty()) return;
            m_stack.back()->on_ui_render(m_ctx);
        }

        void on_event(const SDL_Event& event) {
            if (m_stack.empty()) return;
            m_stack.back()->on_event(event, m_ctx);
        }

        // ---- Debug ----
        size_t depth() const { return m_stack.size(); }

        const char* top_scene_name() const {
            return m_stack.empty() ? "(empty)" : typeid(*m_stack.back()).name();
        }

    private:
        void apply_transition(Scene& current) {
            const auto type = current.m_pending_type;
            auto pending = std::move(current.m_pending_scene);
            current.m_pending_type = SceneTransitionType::None;

            switch (type) {
            case SceneTransitionType::Push:
                if (pending) {
                    Logger::info("SceneManager: push");
                    // Текущая сцена на паузе — on_exit НЕ вызывается
                    pending->on_enter(m_ctx);
                    m_stack.push_back(std::move(pending));
                }
                break;

            case SceneTransitionType::Pop:
                if (m_stack.size() > 1) {
                    Logger::info("SceneManager: pop");
                    m_stack.back()->on_exit(m_ctx);
                    m_stack.pop_back();
                    // Возвращаемся к предыдущей — она уже в стеке, on_enter НЕ нужен
                }
                else {
                    Logger::warning("SceneManager: pop on bottom scene (ignored)");
                }
                break;

            case SceneTransitionType::Replace:
                if (pending) {
                    Logger::info("SceneManager: replace");
                    m_stack.back()->on_exit(m_ctx);
                    m_stack.pop_back();
                    pending->on_enter(m_ctx);
                    m_stack.push_back(std::move(pending));
                }
                break;

            case SceneTransitionType::None:
                break;
            }
        }

        SceneContext& m_ctx;
        std::vector<std::unique_ptr<Scene>> m_stack;
    };


    // ---- Out-of-line: Scene::request_* (после SceneManager) ----
    inline void Scene::request_push(std::unique_ptr<Scene> next) {
        m_pending_type = SceneTransitionType::Push;
        m_pending_scene = std::move(next);
    }

    inline void Scene::request_pop() {
        m_pending_type = SceneTransitionType::Pop;
        m_pending_scene.reset();
    }

    inline void Scene::request_replace(std::unique_ptr<Scene> next) {
        m_pending_type = SceneTransitionType::Replace;
        m_pending_scene = std::move(next);
    }

} // namespace Solace

    void AnimationStateMachine::add_clip(const std::string& name, AnimationClip clip) {
        if (!m_initialized) return;
        clip.name = name;
        m_clips[name] = std::move(clip);
        Logger::info("ASM: clip added '", name, "' (",
            m_clips[name].duration(), "s, ", m_clips[name].events.size(), " events)");
    }

    void AnimationStateMachine::play(const std::string& name, float fade_time) {
        if (!m_initialized) return;

        auto it = m_clips.find(name);
        if (it == m_clips.end()) {
            Logger::error("ASM: clip not found '", name, "'");
            return;
        }

        // Если уже играем этот клип — не сбрасывать
        if (m_current && m_current->name == name && !m_current->loop) {
            return;
        }

        // Смена клипа с crossfade
        m_previous = m_current;
        m_previous_time = m_current_time;

        m_current = &it->second;
        m_current_name = name;
        m_current_time = 0.0f;
        m_fade_time = fade_time;
        m_fade_timer = fade_time;

        Logger::info("ASM[", m_entity, "]: play '", name,
            "' (fade ", fade_time, "s)");
    }

    void AnimationStateMachine::play_immediate(const std::string& name) {
        play(name, 0.0f);
    }

    bool AnimationStateMachine::is_playing(const std::string& name) const {
        return m_current_name == name;
    }

    bool AnimationStateMachine::is_finished() const {
        if (!m_current) return true;
        if (m_current->loop) return false;
        return m_current_time >= m_current->duration();
    }

    float AnimationStateMachine::fade_progress() const {
        if (m_fade_time <= 0.0f) return 1.0f;
        return 1.0f - (m_fade_timer / m_fade_time);
    }

    bool AnimationStateMachine::is_hitbox_active() const {
        if (!m_current) return false;
        const int frame = m_current->frame_at(m_current_time);
        for (const auto& ev : m_current->events) {
            if (ev.name == "hit" && ev.frame == frame) return true;
        }
        return false;
    }

    void AnimationStateMachine::update(float dt) {
        if (!m_initialized || !m_current) return;

        // 1. Продвинуть время
        const float prev_time = m_current_time;
        m_current_time += dt;

        // 2. Loop или завершение
        if (m_current->loop) {
            if (m_current_time >= m_current->duration()) {
                m_current_time = std::fmod(m_current_time, m_current->duration());
            }
        }
        else if (m_current_time >= m_current->duration()) {
            m_current_time = m_current->duration();
            m_bus.queue(AnimationEndedEvent{ m_entity, m_current->name });
        }

        // 3. Emit frame events
        emit_events_between(prev_time, m_current_time);

        // 4. Fade timer
        if (m_fade_timer > 0.0f) {
            m_fade_timer -= dt;
            if (m_fade_timer <= 0.0f) {
                m_fade_timer = 0.0f;
                m_previous = nullptr;  // crossfade завершён
            }
        }

        // 5. Sampling
        sample_clip(0, m_current_time, m_ctx_a, m_locals_a);

        if (m_previous && m_fade_timer > 0.0f) {
            m_previous_time += dt;
            if (m_previous->loop) {
                m_previous_time = std::fmod(m_previous_time, m_previous->duration());
            }
            else {
                m_previous_time = std::min(m_previous_time, m_previous->duration());
            }
            sample_clip(0, m_previous_time, m_ctx_b, m_locals_b);
        }

        // 6. Blending
        blend_and_skin();
    }

    void AnimationStateMachine::emit_events_between(float prev_time, float curr_time) {
        if (!m_current) return;

        const int prev_frame = m_current->frame_at(prev_time);
        const int curr_frame = m_current->frame_at(curr_time);

        // Обрабатываем проход через границу кадра (loop-safe)
        if (curr_frame < prev_frame) {
            // loop перескочил через 0
            for (const auto& ev : m_current->events) {
                if (ev.frame > prev_frame || ev.frame <= curr_frame) {
                    m_bus.queue(AnimationEventTriggered{ m_entity, ev.name });
                }
            }
        }
        else {
            for (const auto& ev : m_current->events) {
                if (ev.frame > prev_frame && ev.frame <= curr_frame) {
                    m_bus.queue(AnimationEventTriggered{ m_entity, ev.name });
                }
            }
        }
    }

    void AnimationStateMachine::sample_clip(
        int /*clip_idx*/,
        float time,
        ozz::animation::SamplingJob::Context& ctx,
        std::vector<ozz::math::SoaTransform>& out_locals)
    {
        if (!m_current || !m_current->animation) return;

        const float ratio = m_current->duration() > 0.0f
            ? time / m_current->duration()
            : 0.0f;

        ozz::animation::SamplingJob job;
        job.animation = m_current->animation;
        job.context = &ctx;
        job.ratio = std::clamp(ratio, 0.0f, 1.0f);
        job.output = ozz::make_span(out_locals);
        job.Run();
    }

    void AnimationStateMachine::blend_and_skin() {
        // Blending
        if (m_previous && m_fade_timer > 0.0f) {
            const float alpha = fade_progress();  // 0 → previous, 1 → current

            ozz::animation::BlendingJob::Layer layers[2];
            layers[0].transform = ozz::make_span(m_locals_b);  // previous
            layers[0].weight = 1.0f - alpha;
            layers[1].transform = ozz::make_span(m_locals_a);  // current
            layers[1].weight = alpha;

            ozz::animation::BlendingJob blend;
            blend.layers = ozz::make_span(layers);
            blend.rest_pose = m_asset->skeleton.joint_rest_poses();
            blend.output = ozz::make_span(m_blended);
            blend.threshold = 0.1f;
            if (!blend.Run()) {
                Logger::error("ASM: blending failed");
                return;
            }
        }
        else {
            // No fade — просто копируем current
            std::copy(m_locals_a.begin(), m_locals_a.end(), m_blended.begin());
        }

        // LocalToModel
        ozz::animation::LocalToModelJob ltm;
        ltm.skeleton = &m_asset->skeleton;
        ltm.input = ozz::make_span(m_blended);
        ltm.output = ozz::make_span(m_models);
        if (!ltm.Run()) {
            Logger::error("ASM: local_to_model failed");
            return;
        }

        // Skinning = Model * InverseBind
        for (size_t i = 0; i < m_models.size(); ++i) {
            m_skinning_matrices[i] = m_models[i] * m_asset->inverse_bind_poses[i];
        }
    }

    const ozz::span<const ozz::math::Float4x4>
        AnimationStateMachine::get_skinning_matrices() const {
        return ozz::make_span(m_skinning_matrices);
    }

    bool Animator::loadSkeleton(const std::string& filename) {
        ozz::io::File file(filename.c_str(), "rb");
        if (!file.opened()) {
            Logger::error("Animator: FAILED to open skeleton file: ", filename);
            return false;
        }
        ozz::io::IArchive archive(&file);
        if (!archive.TestTag<ozz::animation::Skeleton>()) {
            Logger::error("Animator: file is not a valid skeleton: ", filename);
            return false;
        }
        archive >> m_skeleton;
        m_samplingContext.Resize(m_skeleton.num_joints());

        std::vector<ozz::math::SoaTransform> restLocals(m_skeleton.num_soa_joints());
        std::memcpy(restLocals.data(), m_skeleton.joint_rest_poses().data(),
            restLocals.size() * sizeof(ozz::math::SoaTransform));

        std::vector<ozz::math::Float4x4> restModels(m_skeleton.num_joints());
        ozz::animation::LocalToModelJob ltm;
        ltm.skeleton = &m_skeleton;
        ltm.input = ozz::make_span(restLocals);
        ltm.output = ozz::make_span(restModels);
        if (!ltm.Run()) {
            Logger::error("Animator: failed to compute rest pose");
            return false;
        }

        m_inverseBindPoses.resize(m_skeleton.num_joints());
        for (size_t i = 0; i < m_inverseBindPoses.size(); ++i) {
            m_inverseBindPoses[i] = ozz::math::Invert(restModels[i]);
        }

        Logger::info("Animator: skeleton loaded OK. Joints: ", m_skeleton.num_joints());
        return true;
    }

    bool Animator::loadAnimation(const std::string& filename) {
        ozz::io::File file(filename.c_str(), "rb");
        if (!file.opened()) {
            Logger::error("Animator: FAILED to open animation file: ", filename);
            return false;
        }
        ozz::io::IArchive archive(&file);
        if (!archive.TestTag<ozz::animation::Animation>()) {
            Logger::error("Animator: file is not a valid animation: ", filename);
            return false;
        }
        archive >> m_animation;
        m_duration = m_animation.duration();
        m_locals.resize(m_skeleton.num_soa_joints());
        m_models.resize(m_skeleton.num_joints());
        m_skinningMatrices.resize(m_skeleton.num_joints());
        Logger::info("Animator: animation loaded OK. Duration: ", m_duration, "s");
        return true;
    }

    void Animator::update(float deltaTime) {
        if (m_duration <= 0.0f) return;

        m_time += deltaTime;
        if (m_loop) m_time = std::fmod(m_time, m_duration);
        else m_time = std::min(m_time, m_duration);

        float ratio = m_time / m_duration;

        ozz::animation::SamplingJob sampling_job;
        sampling_job.animation = &m_animation;
        sampling_job.context = &m_samplingContext;
        sampling_job.ratio = ratio;
        sampling_job.output = ozz::make_span(m_locals);
        if (!sampling_job.Run()) return;

        // 2. Local -> Model
        ozz::animation::LocalToModelJob ltm_job;
        ltm_job.skeleton = &m_skeleton;
        ltm_job.input = ozz::make_span(m_locals);
        ltm_job.output = ozz::make_span(m_models);
        if (!ltm_job.Run()) {
            Logger::error("ozz: LocalToModel failed!");
            return;
        }

        for (size_t i = 0; i < m_models.size(); ++i) {
            m_skinningMatrices[i] = m_models[i] * m_inverseBindPoses[i];
        }
    }

    std::shared_ptr<ModelAsset> AssetCache::get_model(const std::string& path) {
        auto it = m_models.find(path);
        if (it != m_models.end()) return it->second;

        auto asset = std::make_shared<ModelAsset>();
        asset->path = path;

        try {
            auto mesh_datas = ModelLoader::loadFromFile(path);
            if (mesh_datas.empty()) {
                Logger::error("AssetCache: empty model: ", path);
                return nullptr;
            }

            for (const auto& md : mesh_datas) {
                asset->meshes.push_back(std::make_unique<Mesh>(md));
            }
            build_bone_names(*asset, mesh_datas);
        }
        catch (const std::exception& e) {
            Logger::error("AssetCache: failed to load '", path, "': ", e.what());

        // Skeleton + animations: Phase 0.7 — заглушка.
        // Phase 1.1: загрузка .ozz + маппинг bone_names.
        // Пока asset->has_skeleton() = false.

        m_models.emplace(path, asset);
        Logger::info("AssetCache: loaded '", path, "' (", asset->meshes.size(), " meshes)");

    void processNode(aiNode* node, const aiScene* scene, std::vector<MeshData>& outMeshes, const std::filesystem::path& basePath) {
        for (unsigned int i = 0; i < node->mNumMeshes; i++) {
            aiMesh* aiMeshPtr = scene->mMeshes[node->mMeshes[i]];
            MeshData meshData;

            if (scene->mMaterials && aiMeshPtr->mMaterialIndex < scene->mNumMaterials) {
                aiMaterial* material = scene->mMaterials[aiMeshPtr->mMaterialIndex];
                aiString aiTexPath;
                aiTextureType typesToCheck[] = {
                    aiTextureType_DIFFUSE,
                    aiTextureType_BASE_COLOR,
                    aiTextureType_UNKNOWN,
                    aiTextureType_EMISSIVE,
                    aiTextureType_AMBIENT,
                    aiTextureType_SPECULAR
                };

                Logger::info("--- Material ", aiMeshPtr->mMaterialIndex, " textures ---");
                for (int t = 0; t < aiTextureType_UNKNOWN + 1; t++) {
                    aiTextureType type = static_cast<aiTextureType>(t);
                    UINT count = material->GetTextureCount(type);
                    if (count > 0) {
                        aiString texPath;
                        material->GetTexture(type, 0, &texPath);
                        Logger::info("  Type ", t, ": ", count, " textures, first = '", texPath.C_Str(), "'");
                    }
                }
                bool textureFound = false;
                for (aiTextureType type : typesToCheck) {
                    if (material->GetTextureCount(type) > 0) {
                        if (material->GetTexture(type, 0, &aiTexPath) == AI_SUCCESS) {
                            std::string texPath = aiTexPath.C_Str();
                            std::filesystem::path p(texPath);
                            std::string filename = p.filename().string();
                            const aiTexture* embeddedTex = scene->GetEmbeddedTexture(texPath.c_str());
                            if (embeddedTex) {
                                meshData.hasEmbeddedTexture = true;
                                size_t sizeInBytes = (embeddedTex->mHeight == 0) ? embeddedTex->mWidth : embeddedTex->mWidth * embeddedTex->mHeight * 4;
                                meshData.embeddedBytes.resize(sizeInBytes);
                                std::memcpy(meshData.embeddedBytes.data(), embeddedTex->pcData, sizeInBytes);
                                meshData.diffuseTexturePath = "embedded_" + std::to_string(aiMeshPtr->mMaterialIndex);
                                textureFound = true;
                                break;
                            }
                            else {
                                std::filesystem::path path1 = basePath / "textures" / filename;
                                std::filesystem::path path2 = basePath / filename;
                                if (std::filesystem::exists(path1)) { meshData.diffuseTexturePath = path1.string(); textureFound = true; break; }
                                else if (std::filesystem::exists(path2)) { meshData.diffuseTexturePath = path2.string(); textureFound = true; break; }
                            }
                        }
                    }
                }
                if (!textureFound) {
                    aiColor3D aiColor(1.0f, 1.0f, 1.0f);
                    if (material->Get(AI_MATKEY_COLOR_DIFFUSE, aiColor) == AI_SUCCESS) {
                        meshData.baseColor[0] = aiColor.r;
                        meshData.baseColor[1] = aiColor.g;
                        meshData.baseColor[2] = aiColor.b;
                        Logger::info("ModelLoader: No texture, using material color: R=", aiColor.r, " G=", aiColor.g, " B=", aiColor.b);
                    }
                }
            }



            for (unsigned int v = 0; v < aiMeshPtr->mNumVertices; v++) {
                Vertex vertex{};
                vertex.position[0] = aiMeshPtr->mVertices[v].x;
                vertex.position[1] = aiMeshPtr->mVertices[v].y;
                vertex.position[2] = aiMeshPtr->mVertices[v].z;

                if (aiMeshPtr->HasNormals()) {
                    vertex.normal[0] = aiMeshPtr->mNormals[v].x;
                    vertex.normal[1] = aiMeshPtr->mNormals[v].y;
                    vertex.normal[2] = aiMeshPtr->mNormals[v].z;
                }
                else {
                    vertex.normal[0] = 0.0f; vertex.normal[1] = 1.0f; vertex.normal[2] = 0.0f;
                }

                if (aiMeshPtr->HasTextureCoords(0)) {
                    vertex.uv[0] = aiMeshPtr->mTextureCoords[0][v].x;
                    vertex.uv[1] = aiMeshPtr->mTextureCoords[0][v].y;
                }
                else {
                    vertex.uv[0] = 0.0f; vertex.uv[1] = 0.0f;
                }

                meshData.vertices.push_back(vertex);
            }

            if (aiMeshPtr->HasBones()) {
                Logger::info("  Mesh has ", aiMeshPtr->mNumBones, " bones");

                for (unsigned int b = 0; b < aiMeshPtr->mNumBones; b++) {
                    meshData.boneNames.push_back(aiMeshPtr->mBones[b]->mName.C_Str());
                }

                int vertsWithWeights = 0;
                for (unsigned int b = 0; b < aiMeshPtr->mNumBones; b++) {
                    aiBone* bone = aiMeshPtr->mBones[b];
                    for (unsigned int w = 0; w < bone->mNumWeights; w++) {
                        unsigned int vId = bone->mWeights[w].mVertexId;
                        float weight = bone->mWeights[w].mWeight;
                        if (weight <= 0.0f) continue;
                        if (vId >= meshData.vertices.size()) continue;
                        Vertex& v = meshData.vertices[vId];
                        bool placed = false;
                        for (int s = 0; s < 4; s++) {
                            if (v.boneWeights[s] == 0.0f) {
                                v.boneWeights[s] = weight;
                                v.boneIDs[s] = static_cast<int>(b);
                                placed = true;
                                if (s == 0) vertsWithWeights++;
                                break;
                            }
                        }
                    }
                }
                Logger::info("  Vertices with weights: ", vertsWithWeights, " / ", meshData.vertices.size());
            }
            else {
                Logger::warning("  Mesh has NO bones!");
            }

            for (unsigned int f = 0; f < aiMeshPtr->mNumFaces; f++) {
                aiFace face = aiMeshPtr->mFaces[f];
                for (unsigned int j = 0; j < face.mNumIndices; j++) {
                    meshData.indices.push_back(face.mIndices[j]);
                }
            }

            outMeshes.push_back(meshData);
        }

        for (unsigned int i = 0; i < node->mNumChildren; i++) {
            processNode(node->mChildren[i], scene, outMeshes, basePath);
        }
    }

    std::vector<MeshData> ModelLoader::loadFromFile(const std::string& path) {
        Assimp::Importer importer;

        const aiScene* scene = importer.ReadFile(path,
            aiProcess_Triangulate |
            aiProcess_GenSmoothNormals |
            aiProcess_FlipUVs |
            aiProcess_CalcTangentSpace |
            aiProcess_LimitBoneWeights);

        if (!scene || (scene->mFlags & AI_SCENE_FLAGS_INCOMPLETE) || !scene->mRootNode) {
            throw std::runtime_error("Assimp error loading model (" + path + "): " + importer.GetErrorString());
        }

        std::vector<MeshData> outMeshes;
        std::filesystem::path modelPath(path);
        std::filesystem::path basePath = modelPath.parent_path();

        processNode(scene->mRootNode, scene, outMeshes, basePath);

        Logger::info("ModelLoader: successfully loaded '", path, "' (Meshes found: ", outMeshes.size(), ")");
        return outMeshes;
    }

    MeshData ModelLoader::loadObj(const std::string& path) {
        auto meshes = loadFromFile(path);
        if (meshes.empty()) throw std::runtime_error("ModelLoader: No meshes found in file: " + path);
        return meshes[0];
    }

        m_scenes = std::make_unique<SceneManager>(*m_ctx);

        DebugOverlay::set_context(m_ctx.get(), m_scenes.get());

        // Config hot-reload
        m_bus.subscribe<ConfigReloadRequestEvent>([this](const ConfigReloadRequestEvent&) {
            const bool ok = m_config.reload();
            m_bus.emit(ConfigReloadedEvent{ ok });
            });

        if (m_window && m_window->getSDLWindow()) {
            SDL_Window* sdlWin = m_window->getSDLWindow();
            SDL_SetWindowRelativeMouseMode(sdlWin, true);
            InputSystem::init();

            IMGUI_CHECKVERSION();
            ImGui::CreateContext();
            ImGuiIO& io = ImGui::GetIO();
            io.Fonts->AddFontFromFileTTF("assets/fonts/JetBrainsMono-Regular.ttf", 16.0f);
            io.ConfigFlags |= ImGuiConfigFlags_NavEnableKeyboard;
            ImGui::StyleColorsDark();

            ImGui_ImplSDL3_InitForOpenGL(sdlWin, m_window->GetGLContext());
            ImGui_ImplOpenGL3_Init("#version 430 core");

            DebugOverlay::init();
        }

        onInit();  // ← scene initialization
    }

    void Engine::run() {
        Logger::info("Entering main loop...");
        SDL_Event event;

        while (m_isRunning) {
            m_clock.tick();
            m_registry.process_destroy();       // deferred destroy
            m_bus.processQueue();               // deferred events

            if (!DebugOverlay::is_open()) {
                InputSystem::update();
            }

            while (SDL_PollEvent(&event)) {
                if (event.type == SDL_EVENT_QUIT) m_isRunning = false;

                onEvent(event);                 // Game::onEvent
                m_scenes->on_event(event);       // scene event

                ImGui_ImplSDL3_ProcessEvent(&event);
                if (!DebugOverlay::is_open()) {
                    InputSystem::handleEvent(event);
                }
                if (event.type == SDL_EVENT_WINDOW_RESIZED) {
                    m_window->updateSize(event.window.data1, event.window.data2);
                }
            }

            if (InputSystem::isActionJustPressed(GameAction::ToggleFullscreen)) {
                m_window->toggleFullscreen();
            }
            if (m_window->shouldClose()) m_isRunning = false;

            onUpdate(m_clock.real_dt(), m_clock.gameplay_dt());  // Game::onUpdate
            m_scenes->update(m_clock.gameplay_dt(), m_clock.real_dt());
            m_scenes->process_transitions();

            m_post->beginScene(m_window->width(), m_window->height());
            onRender();                          // Game::onRender (optional)
            m_scenes->render();
            m_post->endScene(m_window->width(), m_window->height());

            onUIRender();                        // Game::onUIRender (ImGui frame)
            m_scenes->ui_render();

            ImGui::Render();
            ImGui_ImplOpenGL3_RenderDrawData(ImGui::GetDrawData());

            m_clock.frame_end();
            m_window->swapBuffers();
        }

        onShutdown();
        m_scenes.reset();                        // ~SceneManager → on_exit всех
    }

    Engine::~Engine() {
        DebugOverlay::shutdown();
        ImGui_ImplOpenGL3_Shutdown();
        ImGui_ImplSDL3_Shutdown();
        ImGui::DestroyContext();
        Logger::info("Engine destroyed.");
    }

    void DebugOverlay::execute_command(const std::string& cmdline) {
        std::stringstream ss(cmdline);
        std::string name;
        ss >> name;
        if (name.empty()) return;

        std::vector<std::string> args;
        std::string arg;
        while (ss >> arg) args.push_back(arg);

        auto it = s_commands.find(name);
        if (it != s_commands.end()) it->second(args);
        else add_log("Unknown command: '" + name + "'");
    }

    void Camera::update(float deltaTime, CameraMode mode, float targetX, float targetY, float targetZ, float yaw, float pitch) {
        if (mode == CameraMode::Noclip) {
            float yawRad = yaw * 3.14159265f / 180.0f;
            float pitchRad = pitch * 3.14159265f / 180.0f;

            float dirX = std::cos(yawRad) * std::cos(pitchRad);
            float dirY = std::sin(pitchRad);
            float dirZ = std::sin(yawRad) * std::cos(pitchRad);

            m_camX = targetX; m_camY = targetY; m_camZ = targetZ;

            m_viewMatrix = Solace::mat4LookAt(
                m_camX, m_camY, m_camZ,
                targetX + dirX, targetY + dirY, targetZ + dirZ,
                0.0f, 1.0f, 0.0f
            );
        }
        else if (mode == CameraMode::Default) {
            float targetEyeX = targetX;
            float targetEyeY = targetY + 3.0f;
            float targetEyeZ = targetZ + 4.5f;

            float smoothSpeed = 5.0f * deltaTime;
            m_camX += (targetEyeX - m_camX) * smoothSpeed;
            m_camY += (targetEyeY - m_camY) * smoothSpeed;
            m_camZ += (targetEyeZ - m_camZ) * smoothSpeed;

            m_viewMatrix = Solace::mat4LookAt(
                m_camX, m_camY, m_camZ,
                targetX, targetY + 1.0f, targetZ,
                0.0f, 1.0f, 0.0f
            );
        }
    }

    GLuint Mesh::getDefaultTextureID() {
        if (s_defaultTextureID == 0) {
            int width, height, channels;

            // Укажи здесь путь к твоей дефолтной текстуре в проекте
            // Например: "assets/textures/default.png" или "assets/default.png"
            std::string defaultTexPath = "assets/textures/missing_texture.jpg";

            unsigned char* imgData = stbi_load(defaultTexPath.c_str(), &width, &height, &channels, 4);

            glGenTextures(1, &s_defaultTextureID);
            glBindTexture(GL_TEXTURE_2D, s_defaultTextureID);

            if (imgData) {
                // Успешно загрузили твой файл текстуры
                glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, width, height, 0, GL_RGBA, GL_UNSIGNED_BYTE, imgData);
                glGenerateMipmap(GL_TEXTURE_2D);

                // Пиксель-арт фильтрация (чтобы не было размытия)
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST_MIPMAP_LINEAR);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);

                stbi_image_free(imgData);
                Logger::info("Mesh: Loaded default texture file: ", defaultTexPath);
            }
            else {
                // Если файл не найден, выводим ошибку и создаем аварийный белый пиксель, чтобы движок не упал
                Logger::error("Mesh: FAILED to load default texture from '", defaultTexPath, "'. Using white fallback.");
                unsigned char whitePixel[] = { 255, 255, 255, 255 };
                glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, 1, 1, 0, GL_RGBA, GL_UNSIGNED_BYTE, whitePixel);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
            }

            glBindTexture(GL_TEXTURE_2D, 0);
        }
        return s_defaultTextureID;
    }

    void Mesh::upload(const MeshData& data) {
        release();
        if (data.vertices.empty() || data.indices.empty()) return;
        m_baseColor[0] = data.baseColor[0];
        m_baseColor[1] = data.baseColor[1];
        m_baseColor[2] = data.baseColor[2];
        m_hasTexture = false;

        bool textureLoaded = false;

        // 1. Пробуем загрузить встроенную текстуру
        if (data.hasEmbeddedTexture && !data.embeddedBytes.empty()) {
            int width, height, channels;
            unsigned char* imgData = stbi_load_from_memory(
                data.embeddedBytes.data(), static_cast<int>(data.embeddedBytes.size()),
                &width, &height, &channels, 4
            );
            if (imgData) {
                glGenTextures(1, &m_textureID);
                glBindTexture(GL_TEXTURE_2D, m_textureID);
                glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, width, height, 0, GL_RGBA, GL_UNSIGNED_BYTE, imgData);
                glGenerateMipmap(GL_TEXTURE_2D);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST_MIPMAP_LINEAR);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
                stbi_image_free(imgData);
                textureLoaded = true;
            }
        }
        // 2. Пробуем загрузить внешнюю текстуру
        else if (!data.diffuseTexturePath.empty()) {
            int width, height, channels;
            unsigned char* imgData = stbi_load(data.diffuseTexturePath.c_str(), &width, &height, &channels, 4);
            if (imgData) {
                glGenTextures(1, &m_textureID);
                glBindTexture(GL_TEXTURE_2D, m_textureID);
                glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, width, height, 0, GL_RGBA, GL_UNSIGNED_BYTE, imgData);
                glGenerateMipmap(GL_TEXTURE_2D);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST_MIPMAP_LINEAR);
                glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
                stbi_image_free(imgData);
                textureLoaded = true;
            }
        }

        if (textureLoaded) {
            m_hasTexture = true;
        }
        else {
            m_hasTexture = false;
        }



        glGenVertexArrays(1, &m_vao);
        glGenBuffers(1, &m_vbo);
        glGenBuffers(1, &m_ebo);
        glBindVertexArray(m_vao);

        glBindBuffer(GL_ARRAY_BUFFER, m_vbo);
        glBufferData(GL_ARRAY_BUFFER, data.vertices.size() * sizeof(Vertex), data.vertices.data(), GL_STATIC_DRAW);

        glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, stride, (void*)offsetof(Vertex, position));
        glEnableVertexAttribArray(1);

        glEnableVertexAttribArray(4);
        glVertexAttribPointer(4, 4, GL_FLOAT, GL_FALSE, stride, (void*)offsetof(Vertex, boneWeights));

        glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, m_ebo);
        glBufferData(GL_ELEMENT_ARRAY_BUFFER, data.indices.size() * sizeof(unsigned int), data.indices.data(), GL_STATIC_DRAW);

        glBindVertexArray(0);
        m_indexCount = static_cast<unsigned int>(data.indices.size());
    }

    void PostProcess::beginScene(int windowW, int windowH) {
        if (windowW != m_lastWindowW || windowH != m_lastWindowH) {
            m_lastWindowW = windowW;
            m_lastWindowH = windowH;


            const int scale = std::max(1, static_cast<int>(std::lround(
                static_cast<double>(windowH) / m_settings.targetHeight)));

            const int newW = (windowW + scale - 1) / scale;
            const int newH = (windowH + scale - 1) / scale;

            if (newW != m_internalW || newH != m_internalH) {
                m_internalW = newW;
                m_internalH = newH;
                recreateTarget();
                Logger::info("PostProcess: internal resolution ",
                    m_internalW, "x", m_internalH, " (pixel ", scale, "x", scale, ")");
            }

            static bool firstFrame = true;
            if (firstFrame) {
                Logger::info("PostProcess: first frame rendered! Drawing quad to screen.");
                firstFrame = false;
            }
        }

        if (m_fbo == 0 || m_colorTex == 0) {
            Logger::error("PostProcess: FBO is not initialized before render!");
            return;
        }

        glBindFramebuffer(GL_FRAMEBUFFER, m_fbo);

        glViewport(0, 0, m_internalW, m_internalH);
        glEnable(GL_DEPTH_TEST);
        glClearColor(0.45f, 0.55f, 0.65f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

        GLenum err = glGetError();
        if (err != GL_NO_ERROR) Logger::error("beginScene: GL error after clear: ", (int)err);
    }

    Shader::Shader(const std::string& vertexPath, const std::string& fragmentPath) {
        std::string vertexCode = readFile(vertexPath);
        std::string fragmentCode = readFile(fragmentPath);

        GLuint vertexShader = compileShader(GL_VERTEX_SHADER, vertexCode);
        GLuint fragmentShader = compileShader(GL_FRAGMENT_SHADER, fragmentCode);

        m_rendererID = glCreateProgram();
        glAttachShader(m_rendererID, vertexShader);
        glAttachShader(m_rendererID, fragmentShader);
        glLinkProgram(m_rendererID);

        GLint success;
        glGetProgramiv(m_rendererID, GL_LINK_STATUS, &success);
        if (!success) {
            GLchar infoLog[512];
            glGetProgramInfoLog(m_rendererID, 512, NULL, infoLog);
            Logger::error("Failed to link shader program! Error: ", infoLog);
        }
        glDeleteShader(vertexShader);
        glDeleteShader(fragmentShader);

        Logger::info("Shader program successfully created (ID: ", m_rendererID, ")");
    }

    std::string Shader::readFile(const std::string& filepath) {
        namespace fs = std::filesystem;

        std::vector<fs::path> candidatePaths = {
            filepath,
            fs::path("..") / filepath,
            fs::path("../..") / filepath,
            fs::path("../../..") / filepath,
            fs::path("../../../..") / filepath
        };

        for (const auto& candidate : candidatePaths) {
            if (fs::exists(candidate) && fs::is_regular_file(candidate)) {
                std::ifstream file;
                file.exceptions(std::ifstream::failbit | std::ifstream::badbit);
                try {
                    file.open(candidate);
                    std::stringstream stream;
                    stream << file.rdbuf();
                    file.close();
                    Logger::debug("Shader successfully loaded from: ", fs::absolute(candidate).string());
                    return stream.str();
                }
                catch (std::ifstream::failure& e) {
                    Logger::warning("Found shader at ", candidate.string(), ", but failed to read it.");
                }
            }
        }
        Logger::error("==================================================");
        Logger::error("FATAL: Failed to find shader file: ", filepath);
        Logger::error("Current Working Directory (CWD): ", fs::current_path().string());
        Logger::error("Check your Visual Studio Working Directory settings!");
        Logger::error("==================================================");

        throw std::runtime_error("Shader file not found: " + filepath);
    }

    GLuint Shader::compileShader(GLenum type, const std::string& source) {
        GLuint shader = glCreateShader(type);

        const char* src = source.c_str();
        glShaderSource(shader, 1, &src, nullptr);

        glCompileShader(shader);

        GLint success;
        glGetShaderiv(shader, GL_COMPILE_STATUS, &success);
        if (!success) {
            GLchar infoLog[512];
            glGetShaderInfoLog(shader, 512, nullptr, infoLog);
            std::string typeStr = (type == GL_VERTEX_SHADER) ? "VERTEX" : "FRAGMENT";
            Logger::error("Failed to compile ", typeStr, " shader! Error: ", infoLog);
        }

        return shader;
    }

        if (state[SDL_SCANCODE_S] || state[SDL_SCANCODE_DOWN])  s_axisValues[(size_t)GameAxis::MoveY] -= 1.0f;

    void InputSystem::handleEvent(const SDL_Event& event) {
        if (event.type == SDL_EVENT_KEY_DOWN && !event.key.repeat) {
            auto it = s_keyToActionMap.find(event.key.scancode);
            if (it != s_keyToActionMap.end()) {
                setActionState(it->second, true);
            }
        }
        else if (event.type == SDL_EVENT_KEY_UP) {
            auto it = s_keyToActionMap.find(event.key.scancode);
            if (it != s_keyToActionMap.end()) {
                setActionState(it->second, false);
            }
        }
        else if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN) {
            auto it = s_mouseButtonToActionMap.find(event.button.button);
            if (it != s_mouseButtonToActionMap.end()) {
                setActionState(it->second, true);
            }
        }
        else if (event.type == SDL_EVENT_MOUSE_BUTTON_UP) {
            auto it = s_mouseButtonToActionMap.find(event.button.button);
            if (it != s_mouseButtonToActionMap.end()) {
                setActionState(it->second, false);
            }
        }
        else if (event.type == SDL_EVENT_MOUSE_MOTION) {
            s_mouseX = event.motion.x;
            s_mouseY = event.motion.y;

            s_axisValues[(size_t)GameAxis::AimX] = s_mouseX;
            s_axisValues[(size_t)GameAxis::AimY] = s_mouseY;

            s_axisValues[(size_t)GameAxis::MouseDeltaX] = event.motion.xrel;
            s_axisValues[(size_t)GameAxis::MouseDeltaY] = event.motion.yrel;
        }
    }

    Mat4 mat4Multiply(const Mat4& a, const Mat4& b) {
        Mat4 r{};
        for (int c = 0; c < 4; ++c) {
            for (int row = 0; row < 4; ++row) {
                r[c * 4 + row] =
                    a[0 * 4 + row] * b[c * 4 + 0] +
                    a[1 * 4 + row] * b[c * 4 + 1] +
                    a[2 * 4 + row] * b[c * 4 + 2] +
                    a[3 * 4 + row] * b[c * 4 + 3];
            }
        }
        return r;
    }

    Mat4 mat4Perspective(float fovYRad, float aspect, float nearZ, float farZ) {
        const float t = 1.0f / std::tan(fovYRad * 0.5f);
        Mat4 m{};
        m[0] = t / aspect;
        m[5] = t;
        m[10] = (farZ + nearZ) / (nearZ - farZ);
        m[11] = -1.0f;
        m[14] = (2.0f * farZ * nearZ) / (nearZ - farZ);
        return m;
    }

    Mat4 mat4LookAt(float ex, float ey, float ez,
        float cx, float cy, float cz,
        float ux, float uy, float uz)
    {
        // (Forward / Z-axis)
        float zx = ex - cx, zy = ey - cy, zz = ez - cz;
        float len = std::sqrt(zx * zx + zy * zy + zz * zz);
        if (len > 0.0f) { zx /= len; zy /= len; zz /= len; }

        // (Right / X-axis) = Up x Forward
        float xx = uy * zz - uz * zy;
        float xy = uz * zx - ux * zz;
        float xz = ux * zy - uy * zx;
        len = std::sqrt(xx * xx + xy * xy + xz * xz);
        if (len > 0.0f) { xx /= len; xy /= len; xz /= len; }

        // (True Up / Y-axis) = Forward x Right
        float yx = zy * xz - zz * xy;
        float yy = zz * xx - zx * xz;
        float yz = zx * xy - zy * xx;

        Mat4 m{};
        // (X-axis / Right)
        m[0] = xx;  m[1] = yx;  m[2] = zx;  m[3] = 0.0f;
        // (Y-axis / Up)
        m[4] = xy;  m[5] = yy;  m[6] = zy;  m[7] = 0.0f;
        // (Z-axis / Forward)
        m[8] = xz;  m[9] = yz;  m[10] = zz; m[11] = 0.0f;
        // (Position / Translation)
        m[12] = -(xx * ex + xy * ey + xz * ez);
        m[13] = -(yx * ex + yy * ey + yz * ez);
        m[14] = -(zx * ex + zy * ey + zz * ez);
        m[15] = 1.0f;

        return m;
    }

#pragma once
#include <string>
#include <nlohmann/json.hpp>

namespace App {

    struct PlayerConfig {
        struct Stats {
            float maxHealth = 100.0f;
            float maxStamina = 100.0f;
            float staminaRegen = 25.0f;
        } stats;

        struct Movement {
            float speed = 5.0f;
            float dashSpeed = 17.5f;
            float dashDuration = 0.25f;
            float dashCooldown = 0.5f;
            float dashStaminaCost = 25.0f;
        } movement;

        struct Animation {
            std::string defaultClip = "idle";
            float fadeTime = 0.2f;
        } animation;

        static bool parse(const nlohmann::json& j, PlayerConfig& out) {
            if (!j.contains("stats") || !j.contains("movement") || !j.contains("animation"))
                return false;

            out.stats.maxHealth = j["stats"].value("maxHealth", out.stats.maxHealth);
            out.stats.maxStamina = j["stats"].value("maxStamina", out.stats.maxStamina);
            out.stats.staminaRegen = j["stats"].value("staminaRegen", out.stats.staminaRegen);

            out.movement.speed = j["movement"].value("speed", out.movement.speed);
            out.movement.dashSpeed = j["movement"].value("dashSpeed", out.movement.dashSpeed);
            out.movement.dashDuration = j["movement"].value("dashDuration", out.movement.dashDuration);
            out.movement.dashCooldown = j["movement"].value("dashCooldown", out.movement.dashCooldown);
            out.movement.dashStaminaCost = j["movement"].value("dashStaminaCost", out.movement.dashStaminaCost);

            out.animation.defaultClip = j["animation"].value("defaultClip", out.animation.defaultClip);
            out.animation.fadeTime = j["animation"].value("fadeTime", out.animation.fadeTime);
            return true;
        }
    };

} // namespace App

    void Game::onInit() {
        // 1. Загрузка graphics (legacy) и применение к PostProcess
        auto settings = Solace::SettingsManager::load();
        Solace::PostSettings pp;
        pp.targetHeight = settings.postProcess.targetHeight;
        pp.paletteBits = settings.postProcess.paletteBits;
        pp.ditherAmount = settings.postProcess.ditherAmount;
        pp.bloomStrength = settings.postProcess.bloomStrength;
        pp.bloomThreshold = settings.postProcess.bloomThreshold;
        getPostProcess()->applySettings(pp);

        // 2. Регистрация gameplay-секций ConfigCache
        config().register_section<PlayerConfig>("player", PlayerConfig::parse);
        if (!config().load("assets/configs")) {
            Solace::Logger::error("Game: failed to load gameplay configs, using defaults");
        }

        // 3. Регистрация консольных команд (упрощённо)
        Solace::DebugOverlay::register_command("fps", [](const std::vector<std::string>& args) {
            if (!args.empty()) {
                Solace::Fps::setVisible(std::stoi(args[0]) > 0);
            }
            });

        // 4. Стартовая сцена
        scenes().push_immediate(std::make_unique<GameplayScene>());

        Solace::Logger::info("Game initialized");
    }

    void Game::onEvent(const SDL_Event& e) {
        if (e.type == SDL_EVENT_KEY_DOWN && !e.key.repeat) {
            if (e.key.scancode == SDL_SCANCODE_F1) {
                Solace::DebugOverlay::toggle_panels();
            }
            if (e.key.scancode == SDL_SCANCODE_GRAVE) {
                Solace::DebugOverlay::toggle();
                SDL_SetWindowRelativeMouseMode(
                    getWindow().getSDLWindow(),
                    !Solace::DebugOverlay::is_open()
                );
            }
            if (e.key.scancode == SDL_SCANCODE_F5) {
                event_bus().emit(Solace::ConfigReloadRequestEvent{});
            }
        }
    }

    void PlayerMovementSystem::update(Player& player, float deltaTime, CameraMode camMode) {
        if (!player.m_isDashing && player.m_stamina < player.m_maxStamina) {
            player.m_stamina += player.m_staminaRegenRate * deltaTime;
            if (player.m_stamina > player.m_maxStamina) player.m_stamina = player.m_maxStamina;
        }

        if (player.m_dashCooldownTimer > 0.0f) player.m_dashCooldownTimer -= deltaTime;
        if (player.m_isDashing) {
            player.m_dashTimer -= deltaTime;
            if (player.m_dashTimer <= 0.0f) player.m_isDashing = false;
        }

        if (Solace::InputSystem::isActionJustPressed(Solace::GameAction::Dash)
            && player.m_stamina >= 25.0f && !player.m_isDashing && player.m_dashCooldownTimer <= 0.0f)
        {
            player.m_isDashing = true;
            player.m_dashTimer = player.m_dashDuration;
            player.m_stamina -= 25.0f;
            player.m_dashCooldownTimer = player.m_dashCooldown;
        }

        float moveX = Solace::InputSystem::getAxis(Solace::GameAxis::MoveX);
        float moveZ = Solace::InputSystem::getAxis(Solace::GameAxis::MoveY);

        float length = std::sqrt(moveX * moveX + moveZ * moveZ);
        if (length > 1.0f) {
            moveX /= length;
            moveZ /= length;
        }

        float currentSpeed = player.m_speed;
        if (player.m_isDashing) {
            currentSpeed *= 3.5f;
        }

        if (camMode == CameraMode::Noclip) {
            float deltaX = Solace::InputSystem::getAxis(Solace::GameAxis::MouseDeltaX);
            float deltaY = Solace::InputSystem::getAxis(Solace::GameAxis::MouseDeltaY);
            float flyY = Solace::InputSystem::getAxis(Solace::GameAxis::FlyY);

            if (deltaX != 0.0f || deltaY != 0.0f) {
                player.m_yaw += deltaX * player.m_mouseSensitivity;
                player.m_pitch -= deltaY * player.m_mouseSensitivity;
                player.m_pitch = std::clamp(player.m_pitch, -89.0f, 89.0f);
            }

            float yawRad = player.m_yaw * 3.14159265f / 180.0f;
            float pitchRad = player.m_pitch * 3.14159265f / 180.0f;

            float forwardX = std::cos(yawRad) * std::cos(pitchRad);
            float forwardY = std::sin(pitchRad);
            float forwardZ = std::sin(yawRad) * std::cos(pitchRad);

            float rightX = -std::sin(yawRad);
            float rightZ = std::cos(yawRad);

            player.m_posX += (forwardX * moveZ + rightX * moveX) * currentSpeed * deltaTime;
            player.m_posY += (forwardY * moveZ + flyY) * currentSpeed * deltaTime;
            player.m_posZ += (forwardZ * moveZ + rightZ * moveX) * currentSpeed * deltaTime;
        }
        else {
            player.m_posX += moveX * currentSpeed * deltaTime;
            player.m_posZ -= moveZ * currentSpeed * deltaTime;
        }
    }

    void PlayerModel::loadModel(const std::string& filePath) {

        }

        try {
            auto meshesData = Solace::ModelLoader::loadFromFile(filePath);
            for (auto& data : meshesData) {
                auto cleanBoneName = [](std::string name) {
                    size_t colon = name.find(':');
                    if (colon != std::string::npos) name = name.substr(colon + 1);

                    size_t pipe = name.rfind('|');
                    if (pipe != std::string::npos) name = name.substr(pipe + 1);
                    if (name.size() > 3 && name[name.size() - 4] == '_' &&
                        isdigit(name[name.size() - 3]) && isdigit(name[name.size() - 2]) &&
                        isdigit(name[name.size() - 1])) {
                        name = name.substr(0, name.size() - 4);
                    }
                    else if (name.size() > 2 && name[name.size() - 3] == '_' &&
                        isdigit(name[name.size() - 2]) && isdigit(name[name.size() - 1])) {
                        name = name.substr(0, name.size() - 3);
                    }

                    std::transform(name.begin(), name.end(), name.begin(), ::tolower);
                    return name;
                    };

                if (!data.boneNames.empty() && m_animator) {
                    const auto& skeleton = m_animator->GetSkeleton();
                    auto jointNames = skeleton.joint_names();

                    Solace::Logger::info("=== Assimp bones (first 5) ===");
                    for (size_t i = 0; i < std::min<size_t>(5, data.boneNames.size()); i++) {
                        Solace::Logger::info("  [", i, "] '", data.boneNames[i], "' -> cleaned '", cleanBoneName(data.boneNames[i]), "'");
                    }

                    Solace::Logger::info("=== ozz joints (first 5) ===");
                    for (size_t i = 0; i < std::min<size_t>(5, jointNames.size()); i++) {
                        if (jointNames[i]) {
                            Solace::Logger::info("  [", i, "] '", jointNames[i], "' -> cleaned '", cleanBoneName(jointNames[i]), "'");
                        }
                    }

                    int mappedWeights = 0;
                    int unmappedWeights = 0;
                    int fullyMappedVerts = 0;
                    int fullyUnmappedVerts = 0;

                    for (auto& v : data.vertices) {
                        float totalW = 0.0f;
                        bool anyMapped = false;

                        for (int s = 0; s < 4; s++) {
                            if (v.boneWeights[s] > 0.0f) {
                                int assimpIdx = v.boneIDs[s];
                                if (assimpIdx >= 0 && assimpIdx < (int)data.boneNames.size()) {
                                    std::string cleanAssimpName = cleanBoneName(data.boneNames[assimpIdx]);
                                    int ozzIdx = -1;

                                    for (size_t j = 0; j < jointNames.size(); j++) {
                                        if (jointNames[j] != nullptr) {
                                            std::string cleanOzzName = cleanBoneName(jointNames[j]);
                                            if (cleanOzzName == cleanAssimpName) {
                                                ozzIdx = (int)j;
                                                break;
                                            }
                                        }
                                    }

                                    if (ozzIdx >= 0) {
                                        v.boneIDs[s] = ozzIdx;
                                        totalW += v.boneWeights[s];
                                        mappedWeights++;
                                        anyMapped = true;
                                    }
                                    else {
                                        if (unmappedWeights < 10) {
                                            Solace::Logger::warning("Unmapped bone: '", data.boneNames[assimpIdx], "' (cleaned: '", cleanAssimpName, "')");
                                        }
                                        v.boneWeights[s] = 0.0f;
                                        v.boneIDs[s] = 0;
                                        unmappedWeights++;
                                    }
                                }
                                else {
                                    v.boneWeights[s] = 0.0f;
                                    v.boneIDs[s] = 0;
                                    unmappedWeights++;
                                }
                            }
                        }

                        if (anyMapped) fullyMappedVerts++; else fullyUnmappedVerts++;

                        if (totalW > 0.01f) {
                            for (int s = 0; s < 4; s++) v.boneWeights[s] /= totalW;
                        }
                        else {
                            for (int s = 0; s < 4; s++) v.boneWeights[s] = 0.0f;
                        }
                    }

                    Solace::Logger::info("Bone mapping: ", mappedWeights, " mapped, ", unmappedWeights, " unmapped");
                    Solace::Logger::info("Verts fully mapped: ", fullyMappedVerts, "/", (int)data.vertices.size());
                }
                m_meshes.push_back(std::make_unique<Solace::Mesh>(data));
            }
            Solace::Logger::info("Player model loaded with SSBO. Meshes: ", m_meshes.size());
        }
        catch (const std::exception& e) {
            Solace::Logger::error("Failed to load player model: ", e.what());
        }

    void PlayerModel::render(Solace::Shader& shader, const Solace::Mat4& viewMatrix, const Solace::Mat4& projMatrix,
        float worldX, float worldY, float worldZ, float yaw)
    {
        if (m_meshes.empty()) return;

        shader.bind();

        Solace::Mat4 translation = Solace::mat4Translate(worldX, worldY, worldZ);
        Solace::Mat4 scaleMat = Solace::mat4Scale(m_scale, m_scale, m_scale);
        Solace::Mat4 modelMatrix = Solace::mat4Multiply(translation, scaleMat);

        shader.setMat4("u_view", viewMatrix.data());
        shader.setMat4("u_proj", projMatrix.data());
        shader.setMat4("u_model", modelMatrix.data());

        if (m_animator) {
            auto bones = m_animator->getBoneMatrices();
            glBindBuffer(GL_SHADER_STORAGE_BUFFER, m_boneSSBO);
            GLsizeiptr requiredSize = static_cast<GLsizeiptr>(bones.size() * sizeof(ozz::math::Float4x4));
            glBufferData(GL_SHADER_STORAGE_BUFFER, requiredSize, bones.data(), GL_DYNAMIC_DRAW);
            glBindBufferBase(GL_SHADER_STORAGE_BUFFER, 0, m_boneSSBO);
            glBindBuffer(GL_SHADER_STORAGE_BUFFER, 0);
        }

        for (const auto& mesh : m_meshes) {
            // Проверяем, есть ли у меша реальная текстура
            if (mesh->getTextureID() != 0) {
                glActiveTexture(GL_TEXTURE0);
                glBindTexture(GL_TEXTURE_2D, mesh->getTextureID());
                shader.setInt("u_textureDiffuse", 0);
                shader.setInt("u_hasTexture", 1);

                // Если текстура есть, белый базовый цвет не будет тонировать её лишним оттенком
                shader.setVec3("u_baseColor", 1.0f, 1.0f, 1.0f);
            }
            else {
                // Если текстуры НЕТ, включаем режим чистой заливки цветом материала
                shader.setInt("u_hasTexture", 0);
                const float* col = mesh->getBaseColor();
                shader.setVec3("u_baseColor", col[0], col[1], col[2]);
            }
            mesh->draw();
        }

        shader.unbind();
    }

    Solace::MeshData WorldGen::buildPillarData(float radius, float height, int segments) {
        Solace::MeshData data;
        const float pi2 = 6.2831853f;
        for (int i = 0; i < segments; ++i) {
            const float a0 = pi2 * static_cast<float>(i) / static_cast<float>(segments);
            const float a1 = pi2 * static_cast<float>(i + 1) / static_cast<float>(segments);
            const float x0 = std::cos(a0) * radius, z0 = std::sin(a0) * radius;
            const float x1 = std::cos(a1) * radius, z1 = std::sin(a1) * radius;
            const float nx = std::cos((a0 + a1) * 0.5f), nz = std::sin((a0 + a1) * 0.5f);

            const unsigned int base = static_cast<unsigned int>(data.vertices.size());

            data.vertices.push_back({ { x0, 0.0f,    z0 },{ nx, 0.0f, nz },{ 0.0f, 0.0f } });
            data.vertices.push_back({ { x1, 0.0f,    z1 },{ nx, 0.0f, nz },{ 1.0f, 0.0f } });
            data.vertices.push_back({ { x1, height,  z1 },{ nx, 0.0f, nz },{ 1.0f, 1.0f } });
            data.vertices.push_back({ { x0, height,  z0 },{ nx, 0.0f, nz },{ 0.0f, 1.0f } });

            data.indices.insert(data.indices.end(),
                { base, base + 2, base + 1,
                  base, base + 3, base + 2 });
        }
        return data;
    }

    bloom += max(texel(pix + vec2(-1.0, -1.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2( 0.0, -1.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2( 1.0, -1.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2(-1.0,  0.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2( 0.0,  0.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2( 1.0,  0.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2(-1.0,  1.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2( 0.0,  1.0)) - u_bloomThreshold, 0.0);
    bloom += max(texel(pix + vec2( 1.0,  1.0)) - u_bloomThreshold, 0.0);

    col += bloom * (u_bloomStrength / 9.0);

    float safePaletteBits = max(u_paletteBits, 5.0); 
    float levels = pow(2.0, safePaletteBits) - 1.0;
    
    float noise  = bayer8(pix) - 0.5;
    col = floor(col * levels + noise * u_dither + 0.5) / levels;

    outColor = vec4(col, 1.0);
}

#version 330 core
out vec4 fragColor;

uniform vec2 u_resolution;
uniform float u_time;

// --- ������� ��� � ��� ---
vec3 hash3(vec3 p) {
    p = vec3(dot(p, vec3(127.1, 311.7, 74.7)),
             dot(p, vec3(269.5, 183.3, 246.1)),
             dot(p, vec3(113.5, 271.9, 124.6)));
    return -1.0 + 2.0 * fract(sin(p) * 43758.5453123);
}

float noise(vec3 p) {
    vec3 i = floor(p);
    vec3 f = fract(p);
    vec3 u = f * f * (3.0 - 2.0 * f);
    return mix(mix(mix(dot(hash3(i + vec3(0,0,0)), f - vec3(0,0,0)),
                       dot(hash3(i + vec3(1,0,0)), f - vec3(1,0,0)), u.x),
                   mix(dot(hash3(i + vec3(0,1,0)), f - vec3(0,1,0)),
                       dot(hash3(i + vec3(1,1,0)), f - vec3(1,1,0)), u.x), u.y),
               mix(mix(dot(hash3(i + vec3(0,0,1)), f - vec3(0,0,1)),
                       dot(hash3(i + vec3(1,0,1)), f - vec3(1,0,1)), u.x),
                   mix(dot(hash3(i + vec3(0,1,1)), f - vec3(0,1,1)),
                       dot(hash3(i + vec3(1,1,1)), f - vec3(1,1,1)), u.x), u.y), u.z);
}

float fbm(vec3 p) {
    float f = 0.0;
    float w = 0.5;
    for (int i = 0; i < 3; i++) { // ��������� �� 3 ����� ��� ������������ ������
        f += w * noise(p);
        p *= 2.0;
        w *= 0.5;
    }
    return f;
}

float ridgedNoise(vec3 p) {
    float f = 0.0;
    float w = 0.5;
    for (int i = 0; i < 3; i++) {
        float n = 1.0 - abs(noise(p));
        n = n * n;
        f += n * w;
        w *= 0.5;
        p *= 2.0;
    }
    return f;
}

// 1. ������ ������� ��������� (���� 1000 FPS ��� �����)
float map(vec3 p) {
    vec3 boxD = abs(p) - vec3(0.9, 1.9, 0.3);
    return length(max(boxD, 0.0)) + min(max(boxD.x, max(boxD.y, boxD.z)), 0.0);
}

vec3 calcNormal(vec3 p) {
    float h = 0.001;
    vec2 k = vec2(1.0, -1.0);
    return normalize(k.xyy * map(p + k.xyy * h) +
                     k.yyx * map(p + k.yyx * h) +
                     k.yxy * map(p + k.yxy * h) +
                     k.xxx * map(p + k.xxx * h));
}

void main() {
    vec2 st = (gl_FragCoord.xy * 2.0 - u_resolution.xy) / min(u_resolution.x, u_resolution.y);

    vec3 ro = u_cameraPos;
    vec3 rd = normalize(vec3(st, 1.0));

    // ������� Raymarching (������� �����)
    float t = 0.0;
    float maxDist = 35.0;

    for (int i = 0; i < 64; i++) { // ����������� ����� �����
        vec3 p = ro + rd * t;
        float d = map(p);
        if (d < 0.001 || t > maxDist) break;
        t += d;
    }

    vec3 color = vec3(0.003, 0.003, 0.008);

    if (t < maxDist) {
        vec3 p = ro + rd * t;
        vec3 baseNormal = calcNormal(p);

        // --- 2. ��������������� �����: ���������� �������� ������ ������ ����� ����� ---
        // ��� ������� ���������� ������ � ������ ��������� ���� ��� � ����� hit, � �� �� ����� ����!

        float slateStrata = abs(sin(p.y * 16.0 + fbm(p * 2.5) * 3.0));
        float fractures = ridgedNoise(p * 3.0);

        // ������������ ������� �� �����������, �������� ������ �������� ������� (Bump/Normal mapping)
        vec3 normal = normalize(baseNormal + vec3(dFdx(slateStrata), dFdy(slateStrata), 0.0) * 0.4);

        // ������� ������� ������ � ��������
        vec3 stoneColor = mix(vec3(0.05, 0.05, 0.07), vec3(0.15, 0.16, 0.18), fbm(p * 4.0));

        // ��������� �������� � ��������
        float quartz = smoothstep(0.7, 0.9, fractures);
        stoneColor = mix(stoneColor, vec3(0.5, 0.6, 0.65), quartz * 0.7);

        // ����� � ����������� ������
        stoneColor *= mix(0.6, 1.0, slateStrata);

        // ���������
        vec3 lightDir = normalize(vec3(0.5, 1.0, -0.4));
        float diff = max(dot(normal, lightDir), 0.0);

        vec3 viewDir = normalize(ro - p);
        vec3 halfDir = normalize(lightDir + viewDir);
        float spec = pow(max(dot(normal, halfDir), 0.0), 32.0);

        vec3 lighting = stoneColor * (diff * 1.5 + 0.1) + vec3(0.25) * spec;

        // �����
        float fog = 1.0 - exp(-0.03 * t * t);
        color = mix(lighting, vec3(0.003, 0.003, 0.008), fog);
    }

    color = pow(color, vec3(1.0 / 2.2));
    fragColor = vec4(color, 1.0);
}
